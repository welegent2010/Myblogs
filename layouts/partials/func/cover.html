{{/* cover ----- Medium 模式：封面就是正文第一张图。 取值顺序： 1. front matter 里的 image 字段 —— 仅在「封面 ≠ 正文首图」时才写，日常不用 2. 正文第一张 markdown 图 ![alt](src) ← 图床链接走这里 3. 正文第一个 img 短代码 {{< img src="" >}} 4. 正文第一个 5. 都没有则返回空 src，由调用方渲染频道配色块 日常写法：Typora 里把图粘贴进正文（PicGo 自动上传图床），第一张即封面。 想换封面就把那张图挪到正文第一位，或临时用 front matter 的 image 覆盖。 返回：dict "src" "external" */}} {{- $p := . -}} {{- $src := "" -}} {{- $external := false -}} {{- with $p.Params.image }}{{ $src = printf "%v" . }}{{ end -}} {{- if not $src -}} {{- $raw := $p.RawContent -}} {{- $md := findRE `!\[[^\]]*\]\(([^)\s]+)` $raw 1 -}} {{- if $md -}} {{- $src = replaceRE `^!\[[^\]]*\]\(` "" (index $md 0) -}} {{- else -}} {{- $shortcode := findRE `\{\{<\s*img[^>]+src="([^"]+)"` $raw 1 -}} {{- if $shortcode -}} {{- $src = replaceRE `^[\s\S]*src="` "" (index $shortcode 0) -}} {{- else -}} {{- $html := findRE `]+src="([^"]+)"` $raw 1 -}} {{- with $html }}{{ $src = replaceRE `^[\s\S]*src="` "" (index . 0) }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $src -}} {{- $src = strings.TrimPrefix "\"" ($src | strings.TrimSuffix "\"") -}} {{- if or (hasPrefix $src "http://") (hasPrefix $src "https://") (hasPrefix $src "//") -}} {{- $external = true -}} {{- end -}} {{- end -}} {{- return dict "src" $src "external" $external -}}
